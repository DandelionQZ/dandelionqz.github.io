일정

JS 코테 함정

  • arr.sort()는 문자열 기준 정렬. 숫자는 꼭 sort((a, b) => a - b).
  • arr.shift()는 O(n). BFS 큐는 head 인덱스로 꺼낸다 (아래 템플릿).
  • 2차원 배열은 Array.from({ length: n }, () => Array(m).fill(0)). Array(n).fill([])는 모든 행이 같은 배열을 공유한다.
  • 재귀 깊이가 1만 단위면 콜스택이 터질 수 있다. 깊은 DFS는 스택으로 바꾼다.
  • 253을 넘는 수는 BigInt. 나머지 연산 문제는 매 단계 % MOD.
  • 키가 문자열이 아니거나 순서가 중요하면 객체 대신 Map. 존재 확인은 Set.
  • 반복문 안 문자열 +=는 느릴 수 있다. 배열에 push 후 join('').
  • 정수 나눗셈은 Math.floor(a / b) (음수면 Math.trunc와 다름 주의).

BFS (격자, head 포인터 큐) 
function bfs(grid, sr, sc) {
  const R = grid.length, C = grid[0].length;
  const dist = Array.from({ length: R }, () => Array(C).fill(-1));
  const q = [[sr, sc]];
  let head = 0;
  dist[sr][sc] = 0;
  const dr = [1, -1, 0, 0], dc = [0, 0, 1, -1];
  while (head < q.length) {
    const [r, c] = q[head++];
    for (let d = 0; d < 4; d++) {
      const nr = r + dr[d], nc = c + dc[d];
      if (nr < 0 || nc < 0 || nr >= R || nc >= C) continue;
      if (grid[nr][nc] === 0 || dist[nr][nc] !== -1) continue;
      dist[nr][nc] = dist[r][c] + 1;
      q.push([nr, nc]);
    }
  }
  return dist;
}
최소 힙 (우선순위 큐) 
class MinHeap {
  constructor() { this.h = []; }
  size() { return this.h.length; }
  peek() { return this.h[0]; }
  push(v) {
    const h = this.h; h.push(v);
    let i = h.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (h[p] <= h[i]) break;
      [h[p], h[i]] = [h[i], h[p]]; i = p;
    }
  }
  pop() {
    const h = this.h, top = h[0], last = h.pop();
    if (h.length) {
      h[0] = last;
      let i = 0;
      while (true) {
        const l = i * 2 + 1, r = l + 1;
        let m = i;
        if (l < h.length && h[l] < h[m]) m = l;
        if (r < h.length && h[r] < h[m]) m = r;
        if (m === i) break;
        [h[m], h[i]] = [h[i], h[m]]; i = m;
      }
    }
    return top;
  }
}

다익스트라 
// graph[u] = [[v, w], ...], 힙은 [dist, node] 비교로 바꾼 MinHeap
function dijkstra(graph, start) {
  const dist = Array(graph.length).fill(Infinity);
  const pq = new MinHeap();
  dist[start] = 0; pq.push([0, start]);
  while (pq.size()) {
    const [d, u] = pq.pop();
    if (d > dist[u]) continue;
    for (const [v, w] of graph[u]) {
      if (d + w < dist[v]) { dist[v] = d + w; pq.push([dist[v], v]); }
    }
  }
  return dist;
}
순열 · 조합 (백트래킹) 
function combinations(arr, k) {
  const res = [], pick = [];
  (function go(start) {
    if (pick.length === k) { res.push([...pick]); return; }
    for (let i = start; i < arr.length; i++) {
      pick.push(arr[i]); go(i + 1); pick.pop();
    }
  })(0);
  return res;
}

function permutations(arr, k) {
  const res = [], pick = [], used = Array(arr.length).fill(false);
  (function go() {
    if (pick.length === k) { res.push([...pick]); return; }
    for (let i = 0; i < arr.length; i++) {
      if (used[i]) continue;
      used[i] = true; pick.push(arr[i]); go(); pick.pop(); used[i] = false;
    }
  })();
  return res;
}
